שיפורים שאני חושב שאפשר לעשות מראש 
- אפשר לעשות שמסמכים רגילים יקלטו בקבוצה ככה שתוכל להעביר מלא, לכתוב אישור וכולם יקלטו ולא יצתרכו לנווט בתפריט ולשלוח אחד אחד 
- אפשר לעשות שאם יש ח.פ של החברה בדוח גיול לא יצתרכו לשלוח, זה נראה לי יחסוך מלא

import requests
import os
import base64
import mimetypes


def get_file_link(filepath, n8n_url, username, password):
    # Read the file's binary content, then encode it into a Base64 text string.
    #  returns
    try:
        with open(filepath, 'rb') as f:
            file_content_binary = f.read()
            file_content_base64 = base64.b64encode(file_content_binary).decode('utf-8')

        # Guess the file's MIME type (e.g., 'application/pdf')
        mime_type, _ = mimetypes.guess_type(filepath)
        if mime_type is None:
            mime_type = 'application/octet-stream'  # Default if type is unknown

        # Create the JSON payload
        payload = {
            "fileName": os.path.basename(filepath),
            "mimeType": mime_type,
            "content": file_content_base64
        }

        # Send the JSON data to the webhook with Basic Auth
        headers = {'Content-Type': 'application/json'}
        response = requests.post(n8n_url, json=payload, headers=headers, auth=(username, password))

        # Check for HTTP errors
        response.raise_for_status()

        return response.json().get("webViewLink")

    except requests.exceptions.RequestException as e:
        print(f"❌ Error: {e}")
        return None
